// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

contract Counter {
    uint256 public count;
    address public immutable owner;

    event CountChanged(uint256 newCount, address indexed caller);

    constructor() { owner = msg.sender; }

    function increment() external {
        count += 1;
        emit CountChanged(count, msg.sender);
    }

    function reset() external {
        require(msg.sender == owner, 'not owner');
        count = 0;
        emit CountChanged(0, msg.sender);
    }
}

4. Solidity 컨트랙트

CoursePoll.sol

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

/// @title CoursePoll
/// @notice 교육용 단일 투표 컨트랙트. 실제 공공선거에 사용하지 마십시오.
contract CoursePoll {
    address public immutable owner;
    string public question;

    string[] private options;
    uint256[] private voteCounts;

    mapping(address => bool) public hasVoted;

    event VoteCast(
        address indexed voter,
        uint256 indexed optionId,
        uint256 newCount
    );

    error AlreadyVoted(address voter);
    error InvalidOption(uint256 optionId);

    constructor(string memory question_, string[] memory options_) {
        require(bytes(question_).length > 0, "Empty question");
        require(options_.length >= 2, "At least two options");

        owner = msg.sender;
        question = question_;
        options = options_;
        voteCounts = new uint256[](options_.length);
    }

    function vote(uint256 optionId) external {
        if (hasVoted[msg.sender]) revert AlreadyVoted(msg.sender);
        if (optionId >= options.length) revert InvalidOption(optionId);

        hasVoted[msg.sender] = true;
        voteCounts[optionId] += 1;

        emit VoteCast(msg.sender, optionId, voteCounts[optionId]);
    }

    function getOptions()
        external
        view
        returns (string[] memory names, uint256[] memory counts)
    {
        return (options, voteCounts);
    }

    function optionCount() external view returns (uint256) {
        return options.length;
    }
}

DemoSpender.sol

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";

/// @title DemoSpender
/// @notice allowance와 transferFrom 위험을 관찰하기 위한 교육용 spender.
contract DemoSpender {
    address public immutable operator;

    event TokensPulled(
        address indexed token,
        address indexed tokenOwner,
        address indexed recipient,
        uint256 amount
    );

    error NotOperator(address caller);
    error InvalidRecipient();
    error TransferFailed();

    constructor() {
        operator = msg.sender;
    }

    function pullTokens(
        address token,
        address tokenOwner,
        address recipient,
        uint256 amount
    ) external {
        if (msg.sender != operator) revert NotOperator(msg.sender);
        if (recipient == address(0)) revert InvalidRecipient();

        bool success = IERC20(token).transferFrom(
            tokenOwner,
            recipient,
            amount
        );
        if (!success) revert TransferFailed();

        emit TokensPulled(token, tokenOwner, recipient, amount);
    }
}

형성평가

1. BrowserProvider의 역할은?

정답 및 해설: MetaMask 같은 EIP-1193 Provider를 ethers의 Provider 인터페이스로 감싸 계정·체인 조회와 Signer 접근을 제공한다.

2. Contract를 Provider에만 연결한 경우와 Signer에 연결한 경우의 차이는?

정답 및 해설: Provider 연결은 주로 읽기 호출, Signer 연결은 상태변경 트랜잭션 서명·전송이 가능하다.

3. vote가 가스를 요구하는 이유는?

정답 및 해설: 컨트랙트 상태를 변경하고 블록에 기록하기 때문이다.

4. approve는 토큰을 즉시 이동시키는가?

정답 및 해설: 아니다. spender가 향후 transferFrom을 사용할 한도를 기록한다.

5. 무제한 승인 후 잔액이 0이면 안전한가?

정답 및 해설: 아니다. 미래에 들어오는 토큰도 allowance 범위에 노출될 수 있다.

6. approve(0)의 목적은?

정답 및 해설: spender의 allowance를 0으로 만들어 향후 transferFrom 권한을 철회하는 것이다.
